// 1. Загружаем модель (делается один раз)
const extractor = await pipeline('feature-extraction', 'Xenova/all-MiniLM-L6-v2');

async function findElementByStep(stepText) {
  // 2. Эмбеддинг шага (например, "Нажми кнопку поиска")
  const stepVector = await extractor(stepText, { pooling: 'mean', normalize: true });

  // 3. Собираем все кнопки и ссылки на странице
  const elements = Array.from(document.querySelectorAll('button, a, input[type="button"]'));
  
  let bestMatch = null;
  let maxScore = -1;

  for (const el of elements) {
    const elText = el.innerText || el.placeholder || el.ariaLabel;
    if (!elText) continue;

    // 4. Эмбеддинг текста элемента
    const elVector = await extractor(elText, { pooling: 'mean', normalize: true });
    
    // 5. Сравнение (Косинусное сходство)
    const score = cosine_similarity(stepVector.data, elVector.data);

    if (score > maxScore) {
      maxScore = score;
      bestMatch = el;
    }
  }

  // 6. Подсвечиваем результат
  if (bestMatch && maxScore > 0.5) {
      bestMatch.style.outline = "3px solid red";
      // bestMatch.click(); // Авто-действие
  }
}

  const expectedVector = await extractor(expectedResultText, { pooling: 'mean', normalize: true });
  
  // Берем все текстовые узлы, которые видны пользователю
  const pageTexts = document.body.innerText.split('\n').filter(t => t.trim().length > 3);
  
  let matchFound = false;
  for (const text of pageTexts) {
    const textVector = await extractor(text, { pooling: 'mean', normalize: true });
    const similarity = cosine_similarity(expectedVector.data, textVector.data);
    
    if (similarity > 0.75) { // Порог уверенности
      console.log(`Подтверждено: "${text}" семантически близко к ожидаемому`);
      matchFound = true;
      break;
    }
  }
  return matchFound;
}
